import hashlib
import json
import subprocess
from concurrent.futures import ThreadPoolExecutor
from pathlib import Path

ROOT = Path('/Users/tommy/Documents/CMC-like-Simulator - Claude')
WORK = Path('/tmp/cmc_plugins_cleanup_2026_10_15')
before = json.loads((WORK / 'before_manifest.json').read_text())
migration = json.loads((WORK / 'binary_migration_receipt.json').read_text())
moved = {e['old_path']: e['new_path'] for e in migration['moved']}
deleted = {e['path'] for e in migration['removed']}
allowed = {'.gitignore', 'AGENTS.md', 'tools/README.md',
           'tools/online_grf_contact/README.md'}
expected = {}
for name in ['runtime_agent_changes.json', 'tools_changes_and_checks.json',
             'trajectory_changes.json']:
    for e in json.loads((WORK / name).read_text())['files']:
        allowed.add(e['path'])
        expected[e['path']] = e['after']['sha256'] if 'after' in e else e['after_sha256']

def inspect(pair):
    old, entry = pair
    new = moved.get(old, old)
    path = ROOT / new
    if old in deleted:
        return {'old': old, 'path': new, 'expected_deleted': True,
                'deletion_correct': not path.exists()}
    if not path.is_file():
        return {'old': old, 'path': new, 'missing': True}
    data = path.read_bytes()
    sha = hashlib.sha256(data).hexdigest()
    return {'old': old, 'path': new, 'bytes': len(data), 'sha256': sha,
            'changed': sha != entry['sha256']}

with ThreadPoolExecutor(max_workers=8) as pool:
    results = list(pool.map(inspect, before['files'].items()))
missing = [r for r in results if r.get('missing')]
changed = [r for r in results if r.get('changed')]
unexpected = [r for r in changed if r['path'] not in allowed]
failed_deletions = [r for r in results if r.get('expected_deleted') and not r['deletion_correct']]
receipt_drift = [r for r in results if r['path'] in expected
                 and r.get('sha256') != expected[r['path']]]
head = subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=ROOT, text=True).strip()
diff = subprocess.run(['git', 'diff', '--check'], cwd=ROOT, capture_output=True, text=True)
installed = sorted(str(p.relative_to(ROOT)) for p in (ROOT / 'plugins').rglob('*')
                   if p.is_file() and (p.suffix.lower() in {'.dylib', '.dll', '.so'}
                                       or '.dylib.' in p.name))
expected_installed = sorted(moved.values())
ignored = {name: subprocess.run(['git', 'check-ignore', '-q', '--', name],
                               cwd=ROOT, capture_output=True).returncode
           for name in expected_installed}
report = {
    'passed': not (missing or unexpected or failed_deletions or receipt_drift)
              and head == before['head'] and diff.returncode == 0
              and installed == expected_installed and all(code == 0 for code in ignored.values()),
    'files_checked': len(results), 'bytes_before': before['bytes_checked'],
    'expected_deletions': len(deleted),
    'byte_identical_files': sum(not r.get('changed') and not r.get('missing')
                                and not r.get('expected_deleted') for r in results),
    'changed_count': len(changed), 'changed_files': changed,
    'missing_files': missing, 'unexpected_changes': unexpected,
    'failed_deletions': failed_deletions, 'agent_receipt_drift': receipt_drift,
    'installed_binaries': installed, 'binary_ignore_check_exit_codes': ignored,
    'head_before': before['head'], 'head_after': head,
    'diff_check_exit_code': diff.returncode,
    'diff_check_output': diff.stdout + diff.stderr,
    'scope': 'All present tracked files plus all model, plugin, previous user/cleanup report files and selected local metadata, captured before migration.',
}
(WORK / 'final_repository_integrity.json').write_text(json.dumps(report, indent=2) + '\n')
print(json.dumps({k: v for k, v in report.items() if k != 'changed_files'}, indent=2))
raise SystemExit(0 if report['passed'] else 1)
